cmake_minimum_required(VERSION 3.8)
# 고규범 신규 패키지 (2026-10-21): common observation interfaces and monitor.
project(vehicle_recorder)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosbag2_transport REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/CommandEvent.msg" "msg/StopStatus.msg"
  DEPENDENCIES std_msgs builtin_interfaces)
add_executable(record_backend src/record_backend.cpp)
target_compile_features(record_backend PUBLIC cxx_std_17)
ament_target_dependencies(record_backend rclcpp rosbag2_cpp rosbag2_transport yaml_cpp_vendor)
target_link_libraries(record_backend yaml-cpp)
install(TARGETS record_backend DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/stop_observer scripts/record DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/stop_observation.py scripts/recorder_node.py scripts/run_recorder.py scripts/record_terminal.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME test_stop_observation COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_stop_observation.py)
  add_test(NAME test_recorder COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_recorder.py)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
